// CPU-only measurement adapter. No model load or backend initialization.
// Tokenizer construction: src/vllm/entrypoints/model_loader.cpp:3088.
// Encoding policy: examples/bench/bench_core.h:324.
#include <fstream>
#include <iostream>
#include <stdexcept>
#include <string>
#include <vector>
#include <nlohmann/json.hpp>
#include "vllm/model_executor/model_loader/gguf_reader.h"
#include "vllm/tokenizer/tokenizer.h"

int main(int argc, char** argv) {
  try {
    if (argc != 3) throw std::runtime_error("usage: native_prompt_ids MODEL.gguf WORKLOAD.json");
    std::ifstream input(argv[2]);
    if (!input) throw std::runtime_error("cannot open workload");
    nlohmann::json workload;
    input >> workload;
    if (!workload.is_array()) throw std::runtime_error("workload must be an array");
    // Same first-turn extraction as bench_core.h:561-585, without truncation.
    std::vector<std::string> prompts;
    for (const auto& entry : workload) {
      if (!entry.contains("conversations") ||
          !entry["conversations"].is_array() ||
          entry["conversations"].empty()) continue;
      const auto& turn = entry["conversations"][0];
      if (!turn.contains("value") || !turn["value"].is_string()) continue;
      prompts.push_back(turn["value"].get<std::string>());
    }
    if (prompts.empty() || prompts.size() != workload.size())
      throw std::runtime_error("invalid or empty workload entries");
    auto gguf = vllm::GgufFile::Open(argv[1]);
    auto tokenizer = vllm::tok::Tokenizer::FromGguf(gguf);
    std::vector<std::vector<int32_t>> ids;
    std::vector<size_t> counts;
    size_t total = 0;
    for (const auto& prompt : prompts) {
      ids.push_back(tokenizer.EncodeWithSpecialTokens(prompt));
      counts.push_back(ids.back().size());
      total += ids.back().size();
    }
    nlohmann::json receipt = {
        {"cpu_only", true}, {"model_path", argv[1]},
        {"workload_path", argv[2]}, {"prompts", prompts},
        {"prompt_token_ids", ids}, {"prompt_token_counts", counts},
        {"total_prompt_tokens", total},
        {"construction", "GgufFile::Open -> Tokenizer::FromGguf"},
        {"tokenization", "EncodeWithSpecialTokens(raw first conversation value)"},
        {"vocab_size", tokenizer.VocabSize()},
        {"bos_id", tokenizer.BosId()}, {"eos_id", tokenizer.EosId()}};
    std::cout << receipt.dump(2) << '\n';
  } catch (const std::exception& error) {
    std::cerr << error.what() << '\n';
    return 1;
  }
}
